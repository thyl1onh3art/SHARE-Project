# SHARE Project API Test Commands
# Copy and paste these commands into your terminal or use them in Postman

# 1. Health Check
curl -X GET http://localhost:5000/health

# 2. Register User
curl -X POST http://localhost:5000/api/users/register \
  -H "Content-Type: application/json" \
  -d '{
    "name": "John Doe",
    "email": "john@example.com",
    "password": "SecurePass123",
    "age": 25,
    "interests": ["finance", "technology"]
  }'

# 3. Login User (copy the token from response)
curl -X POST http://localhost:5000/api/users/login \
  -H "Content-Type: application/json" \
  -d '{
    "email": "john@example.com",
    "password": "SecurePass123"
  }'

# 4. Get User Profile (replace YOUR_TOKEN with actual token)
curl -X GET http://localhost:5000/api/users/me \
  -H "Authorization: Bearer YOUR_TOKEN"

# 5. Create Financial Record (replace YOUR_TOKEN with actual token)
curl -X POST http://localhost:5000/api/finance \
  -H "Content-Type: application/json" \
  -H "Authorization: Bearer YOUR_TOKEN" \
  -d '{
    "type": "output",
    "amount": 50.00,
    "description": "Grocery shopping",
    "date": "2024-01-15T10:00:00Z"
  }'

# 6. Get Financial Records (replace YOUR_TOKEN with actual token)
curl -X GET http://localhost:5000/api/finance \
  -H "Authorization: Bearer YOUR_TOKEN"

# 7. Create Shared Account (replace YOUR_TOKEN with actual token)
curl -X POST http://localhost:5000/api/shared-accounts \
  -H "Content-Type: application/json" \
  -H "Authorization: Bearer YOUR_TOKEN" \
  -d '{
    "name": "Roommate Expenses",
    "memberIds": []
  }'

# 8. Get Shared Accounts (replace YOUR_TOKEN with actual token)
curl -X GET http://localhost:5000/api/shared-accounts \
  -H "Authorization: Bearer YOUR_TOKEN"

# 9. Send Invitation (replace YOUR_TOKEN and ACCOUNT_ID with actual values)
curl -X POST http://localhost:5000/api/invites/send \
  -H "Content-Type: application/json" \
  -H "Authorization: Bearer YOUR_TOKEN" \
  -d '{
    "sharedAccountId": "ACCOUNT_ID",
    "recipientEmail": "friend@example.com",
    "recipientPhone": "+1234567890"
  }'

# 10. List Invitations (replace YOUR_TOKEN with actual token)
curl -X GET "http://localhost:5000/api/invites/list?status=pending" \
  -H "Authorization: Bearer YOUR_TOKEN"
